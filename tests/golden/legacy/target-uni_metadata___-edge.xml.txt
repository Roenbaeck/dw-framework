--------------------------------------------------------------------------
-- Procedure: MergeTable__StageMerge
-- Source: StageMerge
-- Target: MergeTable (merge)
--
-- Map: s1 to Key (as natural key)
-- Map: s2 to StaticField [D] (as static)
-- Map: s3 to ValueField [D]
-- Map: s4 to HistoryField (as history)
-- Map: s5 to OpinionField (as opinion)
-- Map: s6 to Metadata (as metadata)
--
-- [D] marks deletable attributes
-- Generated: <timestamp> by test-user
-- From: test-computer in the test-domain domain
--------------------------------------------------------------------------
CREATE PROCEDURE [dbo].[MergeTable__StageMerge] (
    @agentJobId uniqueidentifier = null,
    @agentStepId smallint = null
)
AS
BEGIN
SET NOCOUNT ON;
DECLARE @inserts int;
DECLARE @updates int;
DECLARE @deletes int;
DECLARE @actions TABLE (
    [action] char(1) not null
);
    -- Preparations before the merge -----------------
    select 'before';
    -- Perform the actual merge ----------------------
    MERGE INTO [DW].[dbo].[MergeTable] AS [target]
    USING (
        select s1, s2 from StageMerge
    ) AS [source]
    ON (
        [source].[s1] = [target].[Key]
    AND (
        [source].[s1] > 0
        )
    )
    WHEN NOT MATCHED THEN INSERT (
        [Key],
        [StaticField],
        [ValueField],
        [HistoryField],
        [Metadata]
    )
    VALUES (
        [source].[s1],
        [source].[s2],
        [source].[s3],
        [source].[s4],
        [source].[s6]
    )
    WHEN MATCHED AND (
        ([target].[StaticField] is not null AND [source].[s2] is null)
    OR
        ([target].[StaticField] is null)
    OR
        ([target].[ValueField] is not null AND [source].[s3] is null)
    OR
        ([target].[ValueField] is null OR [source].[s3] <> [target].[ValueField])
    )
    THEN UPDATE
    SET
        [target].[Deletable_Static] = 1,
        [target].[StaticField] =
            CASE
                WHEN ([target].[StaticField] is null)
                THEN [source].[s2]
                ELSE [target].[StaticField]
            END,
        [target].[Deletable_ValueF] = 1,
        [target].[ValueField] = [source].[s3],
        [target].[HistoryField] = [source].[s4],
        [target].[Metadata] = [source].[s6]
    OUTPUT
        LEFT($action, 1) INTO @actions;
    -- workaround for "Protocol error in TDS stream" bug
    SELECT
        @inserts = NULLIF(SUM(CASE WHEN [action] = 'I' THEN 1 ELSE 0 END), 0),
        @updates = NULLIF(SUM(CASE WHEN [action] = 'U' THEN 1 ELSE 0 END), 0),
        @deletes = NULLIF(SUM(CASE WHEN [action] = 'D' THEN 1 ELSE 0 END), 0)
    FROM
        @actions;
    -- Post processing after the merge ---------------
    select 'after';
END
GO
