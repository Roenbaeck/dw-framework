--------------------------------------------------------------------------
-- Procedure: Edge_Wide_BulkInsert
--
-- This procedure performs a BULK INSERT of the given filename into
-- the Edge_Wide_Insert view. The file is loaded row by row
-- into a single column holding the entire row. This ensures that no
-- data is lost when loading.
--
-- This job may called multiple times in a workflow when more than
-- one file matching a given filename pattern is found.
--
-- Generated: <timestamp> by test-user
-- From: test-computer in the test-domain domain
--------------------------------------------------------------------------
CREATE PROCEDURE [dbo].[Edge_Wide_BulkInsert] (
	@filename varchar(2000),
    @lastModified datetime,
    @agentJobId uniqueidentifier = null,
    @agentStepId smallint = null
)
AS
BEGIN
SET NOCOUNT ON;
DECLARE @CO_ID int;
DECLARE @JB_ID int;
DECLARE @inserts int;
DECLARE @updates int;
DECLARE @workId int;
DECLARE @operationsId int;
DECLARE @theErrorLine int;
DECLARE @theErrorMessage varchar(555);
DECLARE @theErrorSeverity int;
DECLARE @theErrorState int;
EXEC Metadata.metadata._WorkStarting
    @configurationName = 'Wide',
    @configurationType = 'Source',
    @WO_ID = @workId OUTPUT,
    @name = 'Edge_Wide_BulkInsert',
    @agentStepId = @agentStepId,
    @agentJobId = @agentJobId
BEGIN TRY
EXEC Metadata.metadata._WorkSourceToTarget
    @OP_ID = @operationsId OUTPUT,
    @WO_ID = @workId,
    @sourceName = @filename,
    @targetName = 'Edge_Wide_Insert',
    @sourceType = 'File',
    @targetType = 'View',
    @sourceCreated = @lastModified,
    @targetCreated = DEFAULT;
    IF Object_ID('dbo.Edge_Wide_Insert', 'V') IS NOT NULL
    BEGIN
    EXEC('
        BULK INSERT [dbo].[Edge_Wide_Insert]
        FROM ''' + @filename + '''
        WITH (
            FORMATFILE = ''<SCRATCH>\edge\formats\wide.xml'',
            TABLOCK
        );
    ');
    SET @inserts = @@ROWCOUNT;
    EXEC Metadata.metadata._WorkSetInserts @workId, @operationsId, @inserts;
    SET @CO_ID = ISNULL((
        SELECT TOP 1
            CO_ID
        FROM
            Metadata.metadata.lCO_Container
        WHERE
            CO_NAM_Container_Name = @filename
        AND
            CO_CRE_Container_Created = @lastModified
    ), 0);
    SET @JB_ID = ISNULL((
        SELECT
            jb.JB_ID
        FROM
            Metadata.metadata.lWO_part_JB_of wojb
        JOIN
            Metadata.metadata.lJB_Job jb
        ON
            jb.JB_ID = wojb.JB_ID_of
        AND
            jb.JB_AID_AID_AgentJobId = @agentJobId
        WHERE
            wojb.WO_ID_part = @workId
    ), 0);
    UPDATE [dbo].[Edge_Wide_RawSplit]
    SET
        metadata_CO_ID =
          case when metadata_CO_ID = 0 then @CO_ID else metadata_CO_ID end,
        metadata_JB_ID =
          case when metadata_JB_ID = 0 then @JB_ID else metadata_JB_ID end;
    SET @updates = @@ROWCOUNT;
    EXEC Metadata.metadata._WorkSetUpdates @workId, @operationsId, @updates;
    END
    EXEC Metadata.metadata._WorkStopping @workId, 'Success';
END TRY
BEGIN CATCH
	SELECT
		@theErrorLine = ERROR_LINE(),
		@theErrorMessage = ERROR_MESSAGE(),
        @theErrorSeverity = ERROR_SEVERITY(),
        @theErrorState = ERROR_STATE();
    EXEC Metadata.metadata._WorkStopping
        @WO_ID = @workId,
        @status = 'Failure',
        @errorLine = @theErrorLine,
        @errorMessage = @theErrorMessage;
    -- Propagate the error
    RAISERROR(
        @theErrorMessage,
        @theErrorSeverity,
        @theErrorState
    );
END CATCH
END

--------------------------------------------------------------------------
-- Procedure: Edge_Wide_CreateSplitViews
--
-- The split view uses a CLR called the Splitter to split rows in the
-- 'raw' table into columns. The Splitter uses a regular expression in
-- which groups indicate which parts should be cut out as columns.
--
-- The view also checks data types and provide the results as well as
-- show the 'raw' cut column value, before any given transformations
-- have taken place.
--
-- If keys are defined, these keys are checked for duplicates and the
-- duplicate number can be found through the view.
--
-- Create: Edge_Wide_Rows_Split
--
-- Generated: <timestamp> by test-user
-- From: test-computer in the test-domain domain
--------------------------------------------------------------------------
CREATE PROCEDURE [dbo].[Edge_Wide_CreateSplitViews] (
    @agentJobId uniqueidentifier = null,
    @agentStepId smallint = null
)
AS
BEGIN
SET NOCOUNT ON;
DECLARE @workId int;
DECLARE @operationsId int;
DECLARE @theErrorLine int;
DECLARE @theErrorMessage varchar(555);
DECLARE @theErrorSeverity int;
DECLARE @theErrorState int;
EXEC Metadata.metadata._WorkStarting
    @configurationName = 'Wide',
    @configurationType = 'Source',
    @WO_ID = @workId OUTPUT,
    @name = 'Edge_Wide_CreateSplitViews',
    @agentStepId = @agentStepId,
    @agentJobId = @agentJobId
BEGIN TRY
    IF Object_ID('dbo.Edge_Wide_Rows_Split', 'V') IS NOT NULL
    DROP VIEW [dbo].[Edge_Wide_Rows_Split];
    EXEC('
    CREATE VIEW [dbo].[Edge_Wide_Rows_Split]
    AS
    SELECT
        t._id,
        t.metadata_CO_ID,
        t.metadata_JB_ID,
        t._timestamp,
        m.[Fixed] as [Fixed_Match],
        t.[Fixed],
        CASE
            WHEN t.[Fixed] is not null AND [dbo].IsType(t.[Fixed], ''varchar(4)'') = 0 THEN ''Conversion to varchar(4) failed''
        END AS [Fixed_Error],
        m.[Quoted] as [Quoted_Match],
        t.[Quoted],
        CASE
            WHEN t.[Quoted] is not null AND [dbo].IsType(t.[Quoted], ''varchar(20)'') = 0 THEN ''Conversion to varchar(20) failed''
        END AS [Quoted_Error],
        m.[Last] as [Last_Match],
        t.[Last],
        CASE
            WHEN t.[Last] is not null AND [dbo].IsType(t.[Last], ''varchar(20)'') = 0 THEN ''Conversion to varchar(20) failed''
        END AS [Last_Error]
    FROM (
        SELECT
            _id,
            metadata_CO_ID,
            metadata_JB_ID,
            _timestamp,
			NULLIF([Fixed], '''') AS [Fixed],
			NULLIF([Quoted], '''') AS [Quoted],
			NULLIF([Last], '''') AS [Last]
        FROM 
            [dbo].[Edge_Wide_RawSplit]
    ) m
    CROSS APPLY (
        SELECT
            _id,
            metadata_CO_ID,
            metadata_JB_ID,
            _timestamp,
            [Fixed] AS [Fixed],
            [Quoted] AS [Quoted],
            [Last] AS [Last]
    ) t;
    ');
    EXEC Metadata.metadata._WorkStopping @workId, 'Success';
END TRY
BEGIN CATCH
	SELECT
		@theErrorLine = ERROR_LINE(),
		@theErrorMessage = ERROR_MESSAGE(),
        @theErrorSeverity = ERROR_SEVERITY(),
        @theErrorState = ERROR_STATE();
    EXEC Metadata.metadata._WorkStopping
        @WO_ID = @workId,
        @status = 'Failure',
        @errorLine = @theErrorLine,
        @errorMessage = @theErrorMessage;
    -- Propagate the error
    RAISERROR(
        @theErrorMessage,
        @theErrorSeverity,
        @theErrorState
    );
END CATCH
END

--
-- This procedure loads data from the 'Split' views into the 'Typed'
-- tables, with the condition that data must conform to the given
-- data types and have no duplicates for defined keys.
--
-- Load: Edge_Wide_Rows_Split into Edge_Wide_Rows_Typed
--
-- Generated: <timestamp> by test-user
-- From: test-computer in the test-domain domain
--------------------------------------------------------------------------
CREATE PROCEDURE [dbo].[Edge_Wide_SplitRawIntoTyped] (
    @agentJobId uniqueidentifier = null,
    @agentStepId smallint = null
)
AS
BEGIN
SET NOCOUNT ON;
DECLARE @JB_ID int;
DECLARE @insert int;
DECLARE @updates int;
DECLARE @workId int;
DECLARE @operationsId int;
DECLARE @theErrorLine int;
DECLARE @theErrorMessage varchar(555);
DECLARE @theErrorSeverity int;
DECLARE @theErrorState int;
EXEC Metadata.metadata._WorkStarting
    @configurationName = 'Wide',
    @configurationType = 'Source',
    @WO_ID = @workId OUTPUT,
    @name = 'Edge_Wide_SplitRawIntoTyped',
    @agentStepId = @agentStepId,
    @agentJobId = @agentJobId
BEGIN TRY
    IF Object_ID('dbo.Edge_Wide_Rows_Typed', 'U') IS NOT NULL
    BEGIN
EXEC Metadata.metadata._WorkSourceToTarget
    @OP_ID = @operationsId OUTPUT,
    @WO_ID = @workId,
    @sourceName = 'Edge_Wide_Rows_Split',
    @targetName = 'Edge_Wide_Rows_Typed',

    AND
        [Last_Error] is null;
    SET @insert = @insert + @@ROWCOUNT;
    EXEC Metadata.metadata._WorkSetInserts @workId, @operationsId, @insert;
    SET @JB_ID = ISNULL((
        SELECT TOP 1
            JB_ID
        FROM
            Metadata.metadata.lJB_Job
        WHERE
            JB_AID_AID_AgentJobId = @agentJobId
    ), 0);
    UPDATE [dbo].[Edge_Wide_Rows_Typed]
    SET
      metadata_JB_ID = @JB_ID
    WHERE
      metadata_JB_ID <> @JB_ID;
    SET @updates = @@ROWCOUNT;
    EXEC Metadata.metadata._WorkSetUpdates @workId, @operationsId, @updates;
    END
    EXEC Metadata.metadata._WorkStopping @workId, 'Success';
END TRY
BEGIN CATCH
	SELECT
		@theErrorLine = ERROR_LINE(),
		@theErrorMessage = ERROR_MESSAGE(),
        @theErrorSeverity = ERROR_SEVERITY(),
        @theErrorState = ERROR_STATE();
    EXEC Metadata.metadata._WorkStopping
        @WO_ID = @workId,
        @status = 'Failure',
        @errorLine = @theErrorLine,
        @errorMessage = @theErrorMessage;
    -- Propagate the error
    RAISERROR(
        @theErrorMessage,
        @theErrorSeverity,
        @theErrorState
    );
END CATCH
END
