--------------------------------------------------------------------------
-- Procedure: MergeTable__StageMerge
-- Source: StageMerge
-- Target: MergeTable (merge)
--
-- Map: s1 to Key (as natural key)
-- Map: s2 to StaticField [D] (as static)
-- Map: s3 to ValueField [D]
-- Map: s4 to HistoryField (as history)
-- Map: s5 to OpinionField (as opinion)
-- Map: s6 to Metadata (as metadata)
--
-- [D] marks deletable attributes
-- Generated: <timestamp> by test-user
-- From: test-computer in the test-domain domain
--------------------------------------------------------------------------
CREATE PROCEDURE [dbo].[MergeTable__StageMerge] (
    @agentJobId uniqueidentifier = null,
    @agentStepId smallint = null
)
AS
BEGIN
SET NOCOUNT ON;
DECLARE @inserts int;
DECLARE @updates int;
DECLARE @deletes int;
DECLARE @actions TABLE (
    [action] char(1) not null
);
DECLARE @workId int;
DECLARE @operationsId int;
DECLARE @theErrorLine int;
DECLARE @theErrorMessage varchar(555);
DECLARE @theErrorSeverity int;
DECLARE @theErrorState int;
EXEC TestMetadata.metadata._WorkStarting
    @configurationName = 'Edge',
    @configurationType = 'Target',
    @WO_ID = @workId OUTPUT,
    @name = 'MergeTable__StageMerge',
    @agentStepId = @agentStepId,
    @agentJobId = @agentJobId
BEGIN TRY
EXEC TestMetadata.metadata._WorkSourceToTarget
    @OP_ID = @operationsId OUTPUT,
    @WO_ID = @workId,
    @sourceName = 'StageMerge',
    @targetName = 'MergeTable',
    @sourceType = 'Table',
    @targetType = 'Table',
    @sourceCreated = DEFAULT,
    @targetCreated = DEFAULT;
    -- Preparations before the merge -----------------
    select 'before';
    -- Perform the actual merge ----------------------
    MERGE INTO [DW].[dbo].[MergeTable] AS [target]
    USING (
        select s1, s2 from StageMerge
    ) AS [source]
    ON (
        [source].[s1] = [target].[Key]
    AND (
        [source].[s1] > 0
        )
    )
    WHEN NOT MATCHED THEN INSERT (
        [Key],
        [StaticField],
        [ValueField],
        [HistoryField],
        [Metadata]
    )
    VALUES (
        [source].[s1],
        [source].[s2],
        [source].[s3],
        [source].[s4],
        [source].[s6]
    )
    WHEN MATCHED AND (
        ([target].[StaticField] is not null AND [source].[s2] is null)
    OR
        ([target].[StaticField] is null)
    OR
        ([target].[ValueField] is not null AND [source].[s3] is null)
    OR
        ([target].[ValueField] is null OR [source].[s3] <> [target].[ValueField])
    )
    THEN UPDATE
    SET
        [target].[Deletable_Static] = 1,
        [target].[StaticField] =
            CASE
                WHEN ([target].[StaticField] is null)
                THEN [source].[s2]
                ELSE [target].[StaticField]
            END,
        [target].[Deletable_ValueF] = 1,
        [target].[ValueField] = [source].[s3],
        [target].[HistoryField] = [source].[s4],
        [target].[Metadata] = [source].[s6]
    OUTPUT
        LEFT($action, 1) INTO @actions;
    -- workaround for "Protocol error in TDS stream" bug
    SELECT
        @inserts = NULLIF(SUM(CASE WHEN [action] = 'I' THEN 1 ELSE 0 END), 0),
        @updates = NULLIF(SUM(CASE WHEN [action] = 'U' THEN 1 ELSE 0 END), 0),
        @deletes = NULLIF(SUM(CASE WHEN [action] = 'D' THEN 1 ELSE 0 END), 0)
    FROM
        @actions;
    EXEC TestMetadata.metadata._WorkSetInserts @workId, @operationsId, @inserts;
    EXEC TestMetadata.metadata._WorkSetUpdates @workId, @operationsId, @updates;
    EXEC TestMetadata.metadata._WorkSetDeletes @workId, @operationsId, @deletes;
    -- Post processing after the merge ---------------
    select 'after';
    EXEC TestMetadata.metadata._WorkStopping @workId, 'Success';
END TRY
BEGIN CATCH
	SELECT
		@theErrorLine = ERROR_LINE(),
		@theErrorMessage = ERROR_MESSAGE(),
        @theErrorSeverity = ERROR_SEVERITY(),
        @theErrorState = ERROR_STATE();
    EXEC TestMetadata.metadata._WorkStopping
        @WO_ID = @workId,
        @status = 'Failure',
        @errorLine = @theErrorLine,
        @errorMessage = @theErrorMessage;
    -- Propagate the error
    RAISERROR(
        @theErrorMessage,
        @theErrorSeverity,
        @theErrorState
    );
END CATCH
END
