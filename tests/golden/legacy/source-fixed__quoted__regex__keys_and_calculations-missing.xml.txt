--------------------------------------------------------------------------
-- Procedure: Edge_Missing$Terminator_BulkInsert
--
-- This procedure performs a BULK INSERT of the given filename into
-- the Edge_Missing$Terminator_Insert view. The file is loaded row by row
-- into a single column holding the entire row. This ensures that no
-- data is lost when loading.
--
-- This job may called multiple times in a workflow when more than
-- one file matching a given filename pattern is found.
--
-- Generated: <timestamp> by test-user
-- From: test-computer in the test-domain domain
--------------------------------------------------------------------------
CREATE PROCEDURE [dbo].[Edge_Missing$Terminator_BulkInsert] (
	@filename varchar(2000),
    @lastModified datetime,
    @agentJobId uniqueidentifier = null,
    @agentStepId smallint = null
)
AS
BEGIN
SET NOCOUNT ON;
DECLARE @CO_ID int;
DECLARE @JB_ID int;
DECLARE @inserts int;
DECLARE @updates int;
    IF Object_ID('dbo.Edge_Missing$Terminator_Insert', 'V') IS NOT NULL
    BEGIN
    EXEC('
        BULK INSERT [dbo].[Edge_Missing$Terminator_Insert]
        FROM ''' + @filename + '''
        WITH (
            FORMATFILE = ''<SCRATCH>\edge\formats\missing.xml'',
            TABLOCK
        );
    ');
    SET @inserts = @@ROWCOUNT;
    SET @CO_ID = 1 + (
        SELECT TOP 1
            metadata_CO_ID
        FROM
            [dbo].[Edge_Missing$Terminator_Raw]
        ORDER BY
            metadata_CO_ID
        DESC
    );
    SET @JB_ID = CHECKSUM(@agentJobId);
    UPDATE [dbo].[Edge_Missing$Terminator_RawSplit]
    SET
        metadata_CO_ID =
          case when metadata_CO_ID = 0 then @CO_ID else metadata_CO_ID end,
        metadata_JB_ID =
          case when metadata_JB_ID = 0 then @JB_ID else metadata_JB_ID end;
    SET @updates = @@ROWCOUNT;
    END
END
GO

--------------------------------------------------------------------------
-- Procedure: Edge_Missing$Terminator_CreateSplitViews
--
-- The split view uses a CLR called the Splitter to split rows in the
-- 'raw' table into columns. The Splitter uses a regular expression in
-- which groups indicate which parts should be cut out as columns.
--
-- The view also checks data types and provide the results as well as
-- show the 'raw' cut column value, before any given transformations
-- have taken place.
--
-- If keys are defined, these keys are checked for duplicates and the
-- duplicate number can be found through the view.
--
-- Create: Edge_Missing$Terminator_Rows_Split
--
-- Generated: <timestamp> by test-user
-- From: test-computer in the test-domain domain
--------------------------------------------------------------------------
CREATE PROCEDURE [dbo].[Edge_Missing$Terminator_CreateSplitViews] (
    @agentJobId uniqueidentifier = null,
    @agentStepId smallint = null
)
AS
BEGIN
SET NOCOUNT ON;
    IF Object_ID('dbo.Edge_Missing$Terminator_Rows_Split', 'V') IS NOT NULL
    DROP VIEW [dbo].[Edge_Missing$Terminator_Rows_Split];
    EXEC('
    CREATE VIEW [dbo].[Edge_Missing$Terminator_Rows_Split]
    AS
    SELECT
        t._id,
        t.metadata_CO_ID,
        t.metadata_JB_ID,
        t._timestamp,
        m.[Only"Name"] as [Only"Name"_Match],
        t.[Only"Name"],
        CASE
            WHEN t.[Only"Name"] is not null AND [dbo].IsType(t.[Only"Name"], ''varchar(10)'') = 0 THEN ''Conversion to varchar(10) failed''
        END AS [Only"Name"_Error]
    FROM (
        SELECT
            _id,
            metadata_CO_ID,
            metadata_JB_ID,
            _timestamp,
			NULLIF([Only"Name"], '''') AS [Only"Name"]
        FROM 
            [dbo].[Edge_Missing$Terminator_RawSplit]
    ) m
    CROSS APPLY (
        SELECT
            _id,
            metadata_CO_ID,
            metadata_JB_ID,
            _timestamp,
            [Only"Name"] AS [Only"Name"]
    ) t;
    ');
END
GO
