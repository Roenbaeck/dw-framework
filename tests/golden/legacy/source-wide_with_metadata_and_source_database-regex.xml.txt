-- This table holds the 'raw' loaded data.
--

-- if its used or is otherwise an incremented number per file.
--
-- _timestamp

--
-- Generated: <timestamp> by test-user
-- From: test-computer in the test-domain domain

--------------------------------------------------------------------------
-- Procedure: Edge_Regex_BulkInsert
--
-- This procedure performs a BULK INSERT of the given filename into
-- the Edge_Regex_Insert view. The file is loaded row by row
-- into a single column holding the entire row. This ensures that no
-- data is lost when loading.
--
-- This job may called multiple times in a workflow when more than
-- one file matching a given filename pattern is found.
--
-- Generated: <timestamp> by test-user
-- From: test-computer in the test-domain domain
--------------------------------------------------------------------------
CREATE PROCEDURE [dbo].[Edge_Regex_BulkInsert] (
	@filename varchar(2000),
    @lastModified datetime,
    @agentJobId uniqueidentifier = null,
    @agentStepId smallint = null
)
AS
BEGIN
SET NOCOUNT ON;
DECLARE @CO_ID int;
DECLARE @JB_ID int;
DECLARE @inserts int;
DECLARE @updates int;
DECLARE @workId int;
DECLARE @operationsId int;
DECLARE @theErrorLine int;
DECLARE @theErrorMessage varchar(555);
DECLARE @theErrorSeverity int;
DECLARE @theErrorState int;
EXEC Metadata.metadata._WorkStarting
    @configurationName = 'Regex',
    @configurationType = 'Source',
    @WO_ID = @workId OUTPUT,
    @name = 'Edge_Regex_BulkInsert',
    @agentStepId = @agentStepId,
    @agentJobId = @agentJobId
BEGIN TRY
EXEC Metadata.metadata._WorkSourceToTarget
    @OP_ID = @operationsId OUTPUT,
    @WO_ID = @workId,
    @sourceName = @filename,
    @targetName = 'Edge_Regex_Insert',
    @sourceType = 'File',
    @targetType = 'View',
    @sourceCreated = @lastModified,
    @targetCreated = DEFAULT;
    IF Object_ID('dbo.Edge_Regex_Insert', 'V') IS NOT NULL
    BEGIN
    EXEC('
        BULK INSERT [dbo].[Edge_Regex_Insert]
        FROM ''' + @filename + '''
        WITH (
            DATAFILETYPE = ''char'',
            FIELDTERMINATOR = ''|'',
            FIRSTROW = 0,
            TABLOCK
        );
    ');
    SET @inserts = @@ROWCOUNT;
    EXEC Metadata.metadata._WorkSetInserts @workId, @operationsId, @inserts;
    SET @CO_ID = ISNULL((
        SELECT TOP 1
            CO_ID
        FROM
            Metadata.metadata.lCO_Container
        WHERE
            CO_NAM_Container_Name = @filename
        AND
            CO_CRE_Container_Created = @lastModified
    ), 0);
    SET @JB_ID = ISNULL((
        SELECT
            jb.JB_ID
        FROM
            Metadata.metadata.lWO_part_JB_of wojb
        JOIN
            Metadata.metadata.lJB_Job jb
        ON
            jb.JB_ID = wojb.JB_ID_of
        AND
            jb.JB_AID_AID_AgentJobId = @agentJobId
        WHERE
            wojb.WO_ID_part = @workId
    ), 0);
    UPDATE [dbo].[Edge_Regex_Raw]
    SET
    metadata_CO_ID =
      case when metadata_CO_ID = 0 then @CO_ID else metadata_CO_ID end,
    metadata_JB_ID =
      case when metadata_JB_ID = 0 then @JB_ID else metadata_JB_ID end;
    SET @updates = @@ROWCOUNT;
    EXEC Metadata.metadata._WorkSetUpdates @workId, @operationsId, @updates;
    END
    EXEC Metadata.metadata._WorkStopping @workId, 'Success';
END TRY
BEGIN CATCH
	SELECT
		@theErrorLine = ERROR_LINE(),
		@theErrorMessage = ERROR_MESSAGE(),
        @theErrorSeverity = ERROR_SEVERITY(),
        @theErrorState = ERROR_STATE();
    EXEC Metadata.metadata._WorkStopping
        @WO_ID = @workId,
        @status = 'Failure',
        @errorLine = @theErrorLine,
        @errorMessage = @theErrorMessage;
    -- Propagate the error
    RAISERROR(
        @theErrorMessage,
        @theErrorSeverity,
        @theErrorState
    );
END CATCH
END
GO

--------------------------------------------------------------------------
-- Procedure: Edge_Regex_CreateSplitViews
--
-- The split view uses a CLR called the Splitter to split rows in the
-- 'raw' table into columns. The Splitter uses a regular expression in
-- which groups indicate which parts should be cut out as columns.
--
-- The view also checks data types and provide the results as well as
-- show the 'raw' cut column value, before any given transformations
-- have taken place.
--
-- If keys are defined, these keys are checked for duplicates and the
-- duplicate number can be found through the view.
--
-- Create: Edge_Regex_Rows_Split
-- Create: Edge_Regex_More_Split
--
-- Generated: <timestamp> by test-user
-- From: test-computer in the test-domain domain
--------------------------------------------------------------------------
CREATE PROCEDURE [dbo].[Edge_Regex_CreateSplitViews] (
    @agentJobId uniqueidentifier = null,
    @agentStepId smallint = null
)
AS
BEGIN
SET NOCOUNT ON;
DECLARE @workId int;
DECLARE @operationsId int;
DECLARE @theErrorLine int;
DECLARE @theErrorMessage varchar(555);
DECLARE @theErrorSeverity int;
DECLARE @theErrorState int;
EXEC Metadata.metadata._WorkStarting
    @configurationName = 'Regex',
    @configurationType = 'Source',
    @WO_ID = @workId OUTPUT,
    @name = 'Edge_Regex_CreateSplitViews',
    @agentStepId = @agentStepId,
    @agentJobId = @agentJobId
BEGIN TRY
    IF Object_ID('dbo.Edge_Regex_Rows_Split', 'V') IS NOT NULL
    DROP VIEW [dbo].[Edge_Regex_Rows_Split];
    EXEC('
    CREATE VIEW [dbo].[Edge_Regex_Rows_Split]
    AS
    SELECT
        t._id,
        t.metadata_CO_ID,
        t.metadata_JB_ID,
        t._timestamp,
        m.[Fixed] as [Fixed_Match],
        t.[Fixed],
        CASE
            WHEN t.[Fixed] is null THEN ''Null value not allowed''
            WHEN t.[Fixed] is not null AND [dbo].IsType(t.[Fixed], ''varchar(3)'') = 0 THEN ''Conversion to varchar(3) failed''
        END AS [Fixed_Error],
        m.[Delimited] as [Delimited_Match],
        t.[Delimited],
        CASE
            WHEN t.[Delimited] is null THEN ''Null value not allowed''
            WHEN t.[Delimited] is not null AND [dbo].IsType(t.[Delimited], ''varchar(20)'') = 0 THEN ''Conversion to varchar(20) failed''
        END AS [Delimited_Error],
        m.[Pattern] as [Pattern_Match],
        t.[Pattern],
        CASE
            WHEN t.[Pattern] is null THEN ''Null value not allowed''
            WHEN t.[Pattern] is not null AND [dbo].IsType(t.[Pattern], ''varchar(20)'') = 0 THEN ''Conversion to varchar(20) failed''
        END AS [Pattern_Error],
        m.[Rest] as [Rest_Match],
        t.[Rest],
        CASE
            WHEN t.[Rest] is not null AND [dbo].IsType(t.[Rest], ''varchar(20)'') = 0 THEN ''Conversion to varchar(20) failed''
        END AS [Rest_Error],
        ROW_NUMBER() OVER (
            PARTITION BY
                t.[Fixed],
                t.[Delimited]
            ORDER BY
                t._id
        ) - 1 as Primary_Duplicate,
        ROW_NUMBER() OVER (
            PARTITION BY
                t.[Pattern]
            ORDER BY
                t._id
        ) - 1 as Unique_Duplicate
    FROM (
        SELECT TOP(2147483647)
            *
        FROM (
        SELECT ''O''''Brien'' AS [row]
        ) src
        ORDER BY
            _id ASC
    ) forcedMaterializationTrick
    CROSS APPLY (
		SELECT
			NULLIF([1], ''x''y'') AS [Fixed],
			NULLIF([2], ''n''o'') AS [Delimited],
			NULLIF([3], ''n''o'') AS [Pattern],
			NULLIF([4], ''n''o'') AS [Rest]
		FROM (
            SELECT
                [match],
                ROW_NUMBER() OVER (ORDER BY (SELECT 1)) AS idx
            FROM
                [dbo].Splitter(ISNULL(forcedMaterializationTrick.[row], ''''), N''.{2}(.{3})(.*?)''([a-z'']*)(.*)'')
        ) s
        PIVOT (
            MAX([match]) FOR idx IN ([1], [2], [3], [4])
        ) p
    ) m
    CROSS APPLY (
        SELECT
            _id,
            metadata_CO_ID,
            metadata_JB_ID,
            _timestamp,
            [Fixed] AS [Fixed],
            LTRIM(''a''''b'') AS [Delimited],
            [Pattern] AS [Pattern],
            [Rest] AS [Rest]
    ) t;
    ');
    IF Object_ID('dbo.Edge_Regex_More_Split', 'V') IS NOT NULL
    DROP VIEW [dbo].[Edge_Regex_More_Split];
    EXEC('
    CREATE VIEW [dbo].[Edge_Regex_More_Split]
    AS
    SELECT
        t._id,
        t.metadata_CO_ID,
        t.metadata_JB_ID,
        t._timestamp,
        m.[Last] as [Last_Match],
        t.[Last],
        CASE
            WHEN t.[Last] is not null AND [dbo].IsType(t.[Last], ''int'') = 0 THEN ''Conversion to int failed''
        END AS [Last_Error]
    FROM (
        SELECT TOP(2147483647)
            *
        FROM 
            [dbo].[Edge_Regex_Raw] src
        ORDER BY
            _id ASC
    ) forcedMaterializationTrick
    CROSS APPLY (
		SELECT
			NULLIF([1], '''') AS [Last]
		FROM (
            SELECT
                [match],
                ROW_NUMBER() OVER (ORDER BY (SELECT 1)) AS idx
            FROM
                [dbo].Splitter(ISNULL(forcedMaterializationTrick.[row], ''''), N''^x(.*?);'')
        ) s
        PIVOT (
            MAX([match]) FOR idx IN ([1])
        ) p
    ) m
    CROSS APPLY (
        SELECT
            _id,
            metadata_CO_ID,
            metadata_JB_ID,
            _timestamp,
            [Last] AS [Last]
    ) t;
    ');
    EXEC Metadata.metadata._WorkStopping @workId, 'Success';
END TRY
BEGIN CATCH
	SELECT
		@theErrorLine = ERROR_LINE(),
		@theErrorMessage = ERROR_MESSAGE(),
        @theErrorSeverity = ERROR_SEVERITY(),
        @theErrorState = ERROR_STATE();
    EXEC Metadata.metadata._WorkStopping
        @WO_ID = @workId,
        @status = 'Failure',
        @errorLine = @theErrorLine,
        @errorMessage = @theErrorMessage;
    -- Propagate the error
    RAISERROR(
        @theErrorMessage,
        @theErrorSeverity,
        @theErrorState
    );
END CATCH
END

-- Procedure: Edge_Regex_CreateErrorViews
--
-- The created error views can be used to find rows that have errors of
-- the following kinds:
--
-- Type casting errors
-- These errors occur when a value cannot be cast to its designated
-- datatype, for example when trying to cast 'A' to an int.
--
-- Key duplicate errors
-- These errors occur when a primary key is defined and duplicates of
-- that key is found in the tables.
--
-- Create: Edge_Regex_Rows_Error
-- Create: Edge_Regex_More_Error
--
-- Generated: <timestamp> by test-user
-- From: test-computer in the test-domain domain
--------------------------------------------------------------------------
CREATE PROCEDURE [dbo].[Edge_Regex_CreateErrorViews] (
    @agentJobId uniqueidentifier = null,
    @agentStepId smallint = null
)
AS
BEGIN
SET NOCOUNT ON;
DECLARE @workId int;
DECLARE @operationsId int;
DECLARE @theErrorLine int;
DECLARE @theErrorMessage varchar(555);
DECLARE @theErrorSeverity int;
DECLARE @theErrorState int;

    @WO_ID = @workId OUTPUT,

    DROP VIEW [dbo].[Edge_Regex_Rows_Error];
    EXEC('
    CREATE VIEW [dbo].[Edge_Regex_Rows_Error] 
    AS
    SELECT
        *
    FROM
        [dbo].[Edge_Regex_Rows_Split]
    WHERE
        Primary_Duplicate > 0
    OR
        Unique_Duplicate > 0
    OR

    OR

    OR

    OR
        [Rest_Error] is not null;
    ');
    IF Object_ID('dbo.Edge_Regex_More_Error', 'V') IS NOT NULL
    DROP VIEW [dbo].[Edge_Regex_More_Error];
    EXEC('
    CREATE VIEW [dbo].[Edge_Regex_More_Error] 
    AS
    SELECT
        *
    FROM
        [dbo].[Edge_Regex_More_Split]
    WHERE
        [Last_Error] is not null;
    ');
    EXEC Metadata.metadata._WorkStopping @workId, 'Success';
END TRY
BEGIN CATCH

		@theErrorLine = ERROR_LINE(),

--------------------------------------------------------------------------
-- Procedure: Edge_Regex_SplitRawIntoTyped
--
-- This procedure loads data from the 'Split' views into the 'Typed'
-- tables, with the condition that data must conform to the given
-- data types and have no duplicates for defined keys.
--
-- Load: Edge_Regex_Rows_Split into Edge_Regex_Rows_Typed
-- Load: Edge_Regex_More_Split into Edge_Regex_More_Typed
--
-- Generated: <timestamp> by test-user
-- From: test-computer in the test-domain domain
--------------------------------------------------------------------------
CREATE PROCEDURE [dbo].[Edge_Regex_SplitRawIntoTyped] (
    @agentJobId uniqueidentifier = null,
    @agentStepId smallint = null
)
AS
BEGIN
SET NOCOUNT ON;
DECLARE @JB_ID int;
DECLARE @insert int;
DECLARE @updates int;
DECLARE @workId int;
DECLARE @operationsId int;
DECLARE @theErrorLine int;
DECLARE @theErrorMessage varchar(555);
DECLARE @theErrorSeverity int;
DECLARE @theErrorState int;
EXEC Metadata.metadata._WorkStarting
    @configurationName = 'Regex',
    @configurationType = 'Source',
    @WO_ID = @workId OUTPUT,
    @name = 'Edge_Regex_SplitRawIntoTyped',
    @agentStepId = @agentStepId,
    @agentJobId = @agentJobId
BEGIN TRY
    IF Object_ID('dbo.Edge_Regex_Rows_Typed', 'U') IS NOT NULL
    BEGIN
EXEC Metadata.metadata._WorkSourceToTarget
    @OP_ID = @operationsId OUTPUT,
    @WO_ID = @workId,
    @sourceName = 'Edge_Regex_Rows_Split',
    @targetName = 'Edge_Regex_Rows_Typed',
    @sourceType = 'View',
    @targetType = 'Table',
    @sourceCreated = DEFAULT,
    @targetCreated = DEFAULT;
    INSERT INTO [dbo].[Edge_Regex_Rows_Typed] (
        _id,
        metadata_CO_ID,
        metadata_JB_ID,
        [Fixed],
        [Delimited],
        [Pattern],
        [Rest]
    )
    SELECT
        _id,
        metadata_CO_ID,
        metadata_JB_ID,
        [Fixed],
        [Delimited],
        [Pattern],
        [Rest]
    FROM
        [dbo].[Edge_Regex_Rows_Split]
    WHERE
        Primary_Duplicate = 0
    AND
        Unique_Duplicate = 0
    AND
        [Fixed_Error] is null
    AND
        [Delimited_Error] is null
    AND
        [Pattern_Error] is null
    AND
        [Rest_Error] is null;
    SET @insert = @insert + @@ROWCOUNT;
    EXEC Metadata.metadata._WorkSetInserts @workId, @operationsId, @insert;
    SET @JB_ID = ISNULL((
        SELECT TOP 1
            JB_ID
        FROM
            Metadata.metadata.lJB_Job
        WHERE
            JB_AID_AID_AgentJobId = @agentJobId
    ), 0);
    UPDATE [dbo].[Edge_Regex_Rows_Typed]
    SET
      metadata_JB_ID = @JB_ID
    WHERE
      metadata_JB_ID <> @JB_ID;
    SET @updates = @@ROWCOUNT;
    EXEC Metadata.metadata._WorkSetUpdates @workId, @operationsId, @updates;
    END
    IF Object_ID('dbo.Edge_Regex_More_Typed', 'U') IS NOT NULL
    BEGIN
EXEC Metadata.metadata._WorkSourceToTarget
    @OP_ID = @operationsId OUTPUT,
    @WO_ID = @workId,
    @sourceName = 'Edge_Regex_More_Split',
    @targetName = 'Edge_Regex_More_Typed',
    @sourceType = 'View',
    @targetType = 'Table',
    @sourceCreated = DEFAULT,
    @targetCreated = DEFAULT;
    INSERT INTO [dbo].[Edge_Regex_More_Typed] (
        _id,
        metadata_CO_ID,
        metadata_JB_ID,
        [Last]
    )
    SELECT
        _id,
        metadata_CO_ID,
        metadata_JB_ID,
        [Last]
    FROM
        [dbo].[Edge_Regex_More_Split]
    WHERE
        [Last_Error] is null;
    SET @insert = @insert + @@ROWCOUNT;
    EXEC Metadata.metadata._WorkSetInserts @workId, @operationsId, @insert;
    SET @JB_ID = ISNULL((
        SELECT TOP 1
            JB_ID
        FROM
            Metadata.metadata.lJB_Job
        WHERE
            JB_AID_AID_AgentJobId = @agentJobId
    ), 0);
    UPDATE [dbo].[Edge_Regex_More_Typed]
    SET
      metadata_JB_ID = @JB_ID
    WHERE
      metadata_JB_ID <> @JB_ID;
    SET @updates = @@ROWCOUNT;
    EXEC Metadata.metadata._WorkSetUpdates @workId, @operationsId, @updates;
    END
    EXEC Metadata.metadata._WorkStopping @workId, 'Success';
END TRY
BEGIN CATCH
	SELECT
		@theErrorLine = ERROR_LINE(),
		@theErrorMessage = ERROR_MESSAGE(),
        @theErrorSeverity = ERROR_SEVERITY(),
        @theErrorState = ERROR_STATE();
    EXEC Metadata.metadata._WorkStopping
        @WO_ID = @workId,
        @status = 'Failure',
        @errorLine = @theErrorLine,
        @errorMessage = @theErrorMessage;
    -- Propagate the error
    RAISERROR(
        @theErrorMessage,
        @theErrorSeverity,
        @theErrorState
    );
END CATCH
END
GO
